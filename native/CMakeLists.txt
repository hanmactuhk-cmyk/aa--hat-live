cmake_minimum_required(VERSION 3.24)
project(HNAudio VERSION 1.0.0 LANGUAGES C CXX)
if(NOT WIN32)
 message(FATAL_ERROR "Native audio requires Windows x64 and Windows SDK")
endif()
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
include(FetchContent)
FetchContent_Declare(JUCE GIT_REPOSITORY https://github.com/juce-framework/JUCE.git GIT_TAG 8.0.6 GIT_SHALLOW TRUE)
FetchContent_Declare(json URL https://github.com/nlohmann/json/releases/download/v3.11.3/json.tar.xz)
FetchContent_MakeAvailable(JUCE json)
add_executable(hn-audio engine.cpp)
target_compile_definitions(hn-audio PRIVATE _WIN32_WINNT=0x0A00 WINVER=0x0A00 NTDDI_VERSION=0x0A00000A JUCE_STANDALONE_APPLICATION=1 JUCE_MODAL_LOOPS_PERMITTED=1 JUCE_WEB_BROWSER=0 JUCE_USE_CURL=0 JUCE_PLUGINHOST_VST3=1 JUCE_PLUGINHOST_VST=0 JUCE_ASIO=0 JUCE_WASAPI=1 NOMINMAX WIN32_LEAN_AND_MEAN)
target_link_libraries(hn-audio PRIVATE juce::juce_audio_utils juce::juce_dsp nlohmann_json::nlohmann_json ole32 mmdevapi)
if(MSVC)
 target_compile_options(hn-audio PRIVATE /W4 /EHsc)
endif()
